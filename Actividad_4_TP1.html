Consigna:

Realizar un programa que titile 2 leds alternativamente cada 700ms

Ver el circuito usado 

//Este es LO MISMO que el anterior codigo, solo que con un delay mas corto

void setup(){
  pinMode(2, OUTPUT);
  pinMode(3, OUTPUT);
}
void loop(){
  digitalWrite(2, HIGH);
  digitalWrite(3, LOW);
  delay(700);
  digitalWrite(2, LOW);
  digitalWrite(3, HIGH);
  delay(700);
}
