ACTIVIDAD 3 - TP1:

Consigna:

Realizar un programa que titile 2 leds alternativamente

Ver el circuito usado 

void setup(){
  pinMode(2, OUTPUT);      //Denuevo, suministramos energia a los 2 pines
  pinMode(3, OUTPUT);      //El objetivo es, aprobechar que uno NO DEPENDA del suministro de corriente del otro
}
void loop(){
  digitalWrite(2, HIGH);   //OJO, suministramos energia a un pin, y al otro la cortamos
  digitalWrite(3, LOW);    //De esta forma podemos ver el efecto de que titilen
  delay(1000);             //Un tiempo mas grande como pide la actividad
  digitalWrite(2, LOW);    //Acá prendemos el que originalmente estaba apagado (pin 3)
  digitalWrite(3, HIGH);   //Y apagamos al que estaba prendido (pin 2)
  delay(1000);             //Un tiempo para verlos cambiar
}
