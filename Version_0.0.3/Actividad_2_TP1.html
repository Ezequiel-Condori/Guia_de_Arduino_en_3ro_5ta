ACTIVIDAD 2 - TP1:

Consigna:

Realizar un programa que encienda y apague 2 leds al mismo tiempo cada 600ms

Ver el circuito usado 

void setup(){
  pinMode(2, OUTPUT);      //En este caso configuramos 2 pines para salida, asi logramos prender 2 LED's de forma independiente, sin necesidad de que uno dependa del otro
  pinMode(3, OUTPUT);
}
void loop(){
  digitalWrite(2, HIGH);   //Prendemos los 2 a la vez
  digitalWrite(3, HIGH);
  delay(600);              //Damos un tiempo un poco mas corto como pide la actividad
  digitalWrite(2, LOW);    //Apagamos los 2 a la vez
  digitalWrite(3, LOW);
  delay(600);              //Damos un tiempo para ver a los 2 apagarse
}
